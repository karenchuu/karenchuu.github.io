class Solution {
    public void solveSudoku(char[][] board) {
        doSolve(board);
    }

    private boolean doSolve(char[][] board) {
        for (int i = 0; i < board.length; i++) {
            for (int j = 0; j < board[i].length; j++) {
                // 如果有预设数字，不用穷举
                if (board[i][j] != '.') continue;
                for (char num = '1'; num <= '9'; num++) { // try 1 to 9
                    if (isValid(board, i, j, num)) { // 嘗試放進去cell
                        board[i][j] = num;
                                                // 如果找到一个可行解，立即结束
                        if (doSolve(board)) return true;
                        board[i][j] = '.'; // go back
                    }
                }
                // 穷举完 1~9，依然没有找到可行解，此路不通
                // 如果沒有寫這行就會一直for循環，不然就跳不回上一級
                return false;
            }
        }
        return true;
    }

    private boolean isValid(char[][] board, int row, int col, char num) {
        int regionRow = 3 * (row / 3);  //region start row
        int regionCol = 3 * (col / 3);  //region start col
        for (int i = 0; i < 9; i++) {
            if (board[row][i] == num) return false; // check row
            if (board[i][col] == num) return false; // check col
            if (board[regionRow + i / 3][regionCol + i % 3] == num) return false; //check 3*3 block
        }
        return true;
    }
}

      • i%3代表在這個小方塊是第幾欄
