class Solution {
    List<Integer> result = new ArrayList<>();
    public List<Integer> preorder(Node root) {
        preorder_r(root);
        return result;
    }
    private void preorder_r(Node root) {
        if (root == null) return;
        result.add(root.val);
        for (Node node : root.children) {
            preorder_r(node);
        }
    }
}
