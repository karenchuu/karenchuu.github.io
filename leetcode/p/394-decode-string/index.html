class Solution {
    int idx = 0;
    public String decodeString(String s) {
        return decodeHelper(s);
    }

    private String decodeHelper(String s){
        int num = 0;
        StringBuilder sb = new StringBuilder();
        while(idx < s.length()){
            char c = s.charAt(idx++);
            if (Character.isDigit(c)){
                num = num * 10 + (c -'0');
            } else if(c == '[') {
                String temp = decodeHelper(s);
                sb.append(temp.repeat(num));
                num = 0;
            }  else if ( c == ']') {
                break;
            } else {
                sb.append(c);
            }
        }
        return sb.toString();
    }
}
