✍🏽 My Solution

class Solution {
     public boolean search(int[] nums, int target) {
        int n = nums.length;
        int low = 0;
        int high = n-1;
        while (low <= high) {
            int mid = low+(high-low)/2;
            if (nums[mid] == target) {
                return true;
            }
            if (nums[mid] < nums[high]) { // 右邊有序 左邊螺旋有序
                if (target > nums[mid] && target <= nums[high]) { // [mid...right]
                    low = mid + 1;
                } else {
                    high = mid - 1;
                }
            } else if (nums[mid] > nums[high]) { // 左邊有序
                if (target >= nums[low] && target < nums[mid]) { // [left...mid]
                    high = mid - 1;
                } else {
                    low = mid + 1;
                }
            } else {
                high--;
            }
        }
        return false;
    }
}

class Solution {
     public boolean search(int[] nums, int target) {
        int n = nums.length;
        int low = 0;
        int high = n-1;
        while (low <= high) {
            int mid = low+(high-low)/2;
            if (nums[mid] == target) {
                return true;
            }
            if (nums[mid] < nums[high]) { // 右邊有序
                if (target > nums[mid] && target <= nums[high]) { // [mid...right]
                    low = mid + 1;
                } else {
                    high = mid - 1;
                }
            } else if (nums[high] == nums[mid]) {
                high--;
            } else { // 左邊有序
                if (target >= nums[low] && target < nums[mid]) { // [left...mid]
                    high = mid - 1;
                } else {
                    low = mid + 1;
                }
            } 
        }
        return false;
    }
}
