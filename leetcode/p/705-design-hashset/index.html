public class MyHashSet {
    ListNode[] buckets;

    private class ListNode {
        public int key;
        public ListNode next;
        public ListNode(int key) {
            this.key = key;
        }
    }

    /** Initialize your data structure here. */
    public MyHashSet() {
        buckets = new ListNode[13 * 17];
    }

    public void add(int key) {
        int index = key % buckets.length;
        ListNode head = buckets[index]; // 在hashMap[index]的chain裡查看是否已經存在key

        while (head != null && head.key != key) {
            head = head.next;
        }

        if (head != null) {
            return;
        } else {
            ListNode newNode = new ListNode(key);
            newNode.next = buckets[index];
            buckets[index] = newNode;
        }
    }

    // 計算 hashCode(key)=x
    // 找到對應的 index = X % size
    // 在hashMap[index]的chain裡查看是否已經存在key
        // 1.存在 -> 刪除對應的value
        // 2.不存在 -> 什麼都不做
    //size--
    public void remove(int key) {
        int index = key % buckets.length;
        ListNode head = buckets[index];
        if (head == null) {
            return;
        }

        ListNode dummy = new ListNode(0);
        ListNode cur = dummy;
        dummy.next = head;

        while (cur.next != null) {
            if (cur.next.key == key) {
                cur.next = cur.next.next;
                break;
            }
            cur = cur.next;
        }
        buckets[index] = dummy.next;
    }

    /** Returns true if this set contains the specified element */
    public boolean contains(int key) {
        int pos = key % buckets.length;
        ListNode head = buckets[pos];
        if (head == null) {
            return false;
        }

        while (head != null) {
            if (head.key == key) {
                return true;
            }
            head = head.next;
        }

        return false;
    }
}

class MyHashSet {
    private Node[] buckets;
    static final int size = 10000;

    public MyHashSet() {
        buckets = new Node[size];
    }

    private class Node {
        public int key;
        public Node next;
        public Node (int key){
            this.key = key;
            this.next = null;
        }
    }

    public void add(int key) {
        int pos = hash(key);
        Node cur = buckets[pos];
        while (cur != null && cur.key != key) {
            cur = cur.next;
        }
        if (cur != null) {
            return;
        } else {
            Node newNode = new Node(key);
            newNode.next = buckets[pos];
            buckets[pos] =  newNode;
        }
    }

    public boolean contains(int key) {
        int pos = hash(key);
        Node cur = buckets[pos];
        while (cur != null && cur.key != key) {
            cur = cur.next;
        }
        return cur == null ? false : true;
    }

    public void remove(int key) {
        int pos = hash(key);
        Node cur = buckets[pos];

        Node dummy = new Node(0);
        Node prev = dummy;
        dummy.next = cur;

        while (prev.next != null) {
            if (prev.next.key == key) {
                prev.next = prev.next.next;
                break;
            }
            prev = prev.next;
        }
        buckets[pos] = dummy.next;
    }

    private int hash(int key) {
        return key % size;
    }
}

Solution #2 [模擬Java HashSet]

public class MyHashSet {
    class ListNode {
        int key;
        ListNode next;
        ListNode(int key) {
            this.key = key;
        }
    }

    ListNode[] listNodes;

    /** Initialize your data structure here. */
    public MyHashSet() {
        listNodes = new ListNode[13 * 17];
    }

    public void add(int key) {
        int pos = key % listNodes.length;
        ListNode prev = listNodes[pos];
        if (prev == null) {
            listNodes[pos] = new ListNode(key);
            return;
        }

        if (prev.key == key) {
            return;
        }

        while (prev.next != null && prev.next.key != key) {
            prev = prev.next;
        }

        if (prev.next == null) {
            prev.next = new ListNode(key);
        }
    }

    public void remove(int key) {
        int pos = key % listNodes.length;
        ListNode prev = listNodes[pos];
        if (prev == null) {
            return;
        }

        if (prev.key == key) {
            listNodes[pos] = prev.next;
        }

        while (prev.next != null && prev.next.key != key) {
            prev = prev.next;
        }

        if (prev.next != null) {
            prev.next = prev.next.next;
        }
    }

    /** Returns true if this set contains the specified element */
    public boolean contains(int key) {
        int pos = key % listNodes.length;
        ListNode prev = listNodes[pos];
        if (prev == null) {
            return false;
        }

        if (prev.key == key) {
            return true;
        }

        while (prev.next != null && prev.next.key != key) {
            prev = prev.next;
        }

        return prev.next != null;
    }
}
