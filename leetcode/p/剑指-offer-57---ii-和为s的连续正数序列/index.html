class Solution {
    public int[][] findContinuousSequence(int target) {
        List<int[]> result = new ArrayList<>();
        int p = 1, q = 2, sum = 3;
        while (p < q) {
            if (sum == target) {
                int[] arr = new int[q-p+1];
                for (int i = p; i <= q; i++) {
                    arr[i-p] = i;
                }
                result.add(arr);
                sum -= p;
                p++;
                q++;
                sum += q;
            } else if (sum > target) {
                sum -= p;
                p++;
            } else {
                q++;
                sum += q;
            }
        }
        int[][] resultArr = new int[result.size()][];
        for (int i = 0; i < result.size(); i++) {
            resultArr[i] = result.get(i);
        }
        return resultArr;
    }
}
