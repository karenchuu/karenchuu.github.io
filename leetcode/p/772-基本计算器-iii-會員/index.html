class Solution {
    public int calculate(String s) {
        Stack<Integer> nums = new Stack<>();
        Stack<Character> ops = new Stack<>();
        int n = s.length();
        for (int i = 0; i < n; i++) {
            char c = s.charAt(i);
            if (c == ' ') {
                continue;
            }
            if (isDigit(c)) {
                int num = 0;
                while (i < n && isDigit(s.charAt(i))) {
                    num = num * 10 + s.charAt(i) - '0';
                    i++;
                }
                i--;
                nums.push(num);
            } else if (c == '(') {
                ops.push(c);
            } else if (c == ')') {
                while (!ops.isEmpty() && ops.peek() != '(') {
                    fetchAndCal(nums, ops);
                }
                ops.pop(); //彈出 '('
            } else {
                if (ops.isEmpty() || prior(c, ops.peek())) {
                    ops.push(c);
                } else {
                    while (!ops.isEmpty() && !prior(c, ops.peek())) {
                        fetchAndCal(nums, ops);
                    }
                    ops.push(c);
                }
            }
        }
        while (!ops.isEmpty()) {
            fetchAndCal(nums, ops);
        }
        return nums.pop();
    }

    private boolean isDigit(char c) {
        return c >= '0' && c <= '9';
    }

    private boolean prior (char a, char b) {
        if ((a == '*' || a == '/')  && (b == '+' || b == '-')) {
            return true;
        }
        if (b == '(') {
            return true;
        }
        return false;
    }

    private int cal(int num1, int num2, char op) {
        switch (op) {
            case '+': return num1 + num2;
            case '-': return num1 - num2;
            case '*': return num1 * num2;
            case '/': return num1 / num2;
        }
        return -1;
    }

    private void fetchAndCal(Stack<Integer> nums, Stack<Character> ops) {
        int num2 = nums.pop();
        int num1 = nums.pop();
        char op = ops.pop();
        int ret = cal(num1, num2, op);
        nums.push(ret);
    }
}

class Solution {
    public int calculate(String s) {
        Stack<Integer> nums = new Stack<>();
        Stack<Character> ops = new Stack<>();
        int n = s.length(); 
        for (int i = 0; i < n; i++) {
            char c = s.charAt(i);
            if (c == ' ') {
                continue;
            }
            if (isDigit(c)) { // 數字
                // 計算連續數字
                int number = 0;
                while (i < n && isDigit(s.charAt(i))) {
                    number = number*10+(s.charAt(i)-'0');
                    i++;
                }
                i--;  // 这个是为了抵消上面for循环中的i++
                nums.push(number);
            } else if (c == '('){
                ops.push(c);
            } else if (c == ')'){
                while (!ops.isEmpty() && ops.peek() != '(') {
                    fetchAndCal(nums, ops);
                }
                ops.pop(); // 彈出'('
            } else { // 運算符
                if (ops.isEmpty() || prior(c, ops.peek())) {
                    ops.push(c);
                } else {
                    while (!ops.isEmpty() && !prior(c, ops.peek())) {
                        fetchAndCal(nums, ops);
                    }
                    ops.push(c);
                }
            }
        }
        while (!ops.isEmpty()) {
            fetchAndCal(nums, ops);
        }
        return nums.pop();
    }

    private int cal(int num1, int num2, char op) {
        switch(op) {
            case '+': return num1 + num2;
            case '-': return num1 - num2;
            case '*': return num1 * num2;
            case '/': return num1 / num2;
        }
        return -1;
    }

    private boolean isDigit(char c) {
        return c >= '0' && c <= '9';
    }

    private boolean prior(char a, char b) {
        if ((a == '*' || a == '/') && (b == '+' || b == '-')) return true;
        if (b == '(') return true;
        return false;
    }

    private void fetchAndCal(Stack<Integer> nums, Stack<Character> ops) {
        int num2 = nums.pop();
        int num1 = nums.pop();
        char op = ops.pop();
        int ret = cal(num1, num2, op);
        nums.push(ret);
    }
}

Explanation (in my words)

  • How does their solution work?
    • 數字直接入棧
    • 運算符：
      • 棧空或 c > 棧頂運算符，c入棧
      • c ≤ 棧頂運算符，出棧計算，再放進運算符棧頂
    • 如果是 ’(’ 直接入棧
    • 如果是 ’)’ 出棧計算，直到碰到 ’(’ 為止
    • follow up: 如果要再多判斷 { } [ ]，其實跟小括號 ( ) 處理邏輯是一模一樣
  • What is the space/time complexity of their solution?
    • time: O(n)
    • space: O(n)
