744. Find Smallest Letter Greater Than Target

class Solution {
    // 第一個大於給定值target的數
    public char nextGreatestLetter(char[] letters, char target) {
        int n = letters.length;
        int low = 0;
        int high = n-1;
        while (low <= high) {
            int mid = low+(high-low)/2;
            if (letters[mid] > target) {
                if (mid == 0 || letters[mid-1] <= target) {
                    return letters[mid];
                } else {
                    high = mid - 1;
                }
            } else {
                low = mid + 1;
            }
        }
        return letters[0];
    }
}
