class Solution {
    public int smallestDifference(int[] a, int[] b) {
        Arrays.sort(a);
        Arrays.sort(b);
        int m = a.length;
        int n = b.length;
        int i = 0;
        int j = 0;
        long result = Long.MAX_VALUE;
        while (i < m && j < n) {
            if (a[i] < b[j]) {
                result = Math.min(result, (long)b[j]-a[i]);
                i++;
            } else {
                result = Math.min(result, (long)a[i]-b[j]);
                j++;
            }
        }
        return (int)result;
    }
}
