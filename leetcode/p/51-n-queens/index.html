class Solution {
    List<List<String>> res = new ArrayList<>();
    public List<List<String>> solveNQueens(int n) {
        char[][] board = new char[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                board[i][j] = '.';
            }
        }
        backtrack(0, board, n);
        return res;
    }

    private void backtrack(int row, char[][] board, int n) {
        // 結束條件，得到可行解
        if (row == n) {
            List<String> snapshot = new ArrayList<>();
            for (int i = 0; i < n; i++) {
                snapshot.add(new String(board[i]));

            }
            res.add(snapshot);
            return;
        }

        for (int col = 0; col < n; col++) {
            if(isOk(board, n, row, col)) {
                board[row][col] = 'Q';
                backtrack(row+1, board, n);
                board[row][col] = '.';
            }
        }

    }

    private boolean isOk(char[][] board, int n, int row, int col) {
        // 檢查同一欄有沒有衝突
        for (int i = 0; i < row; i++) {
            if (board[i][col] == 'Q') {
                return false;
            }
        }

        // 檢查右上對角線是否有衝突
        int i = row - 1;
        int j = col + 1;
        while (i >= 0 && j < n) {
            if (board[i][j] == 'Q') {
                return false;
            }
            i--;
            j++;
        }
        // 檢查左上對角線是否有衝突
        i = row - 1; 
        j = col - 1;
        while (i >= 0 && j >= 0) {
            if (board[i][j] == 'Q') {
                return false;
            }
            i--;
            j--;
        }
        return true;
    }
}

🧠 Thought Process

  • Why did I choose this particular implementation?
    • 參考王爭
    • 按照题意我们需要检查八个方向是否有其他方向，但实际上我们只需要检查三个方向即可。 因为我们的逻辑是每一行只放一个皇后，所以这个皇后的左边和右边不需要进行检查了。 因为我们是一行一行从上向下放置皇后，所以下方，左下方和右下方不会有皇后（还没放皇后）。 最后我们需要检查的只有上方、左上和右上三个方向。
